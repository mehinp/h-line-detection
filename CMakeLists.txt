cmake_minimum_required(VERSION 3.20)
project(h-line-detection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RTLSDR REQUIRED IMPORTED_TARGET librtlsdr)
pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3f)
add_executable(ingest src/ingest.cpp src/spectrometer.cpp)
target_link_libraries(ingest PRIVATE PkgConfig::RTLSDR PkgConfig::FFTW)
set_target_properties(ingest PROPERTIES BUILD_RPATH "/usr/local/lib")
